1//! Dahlquist Test Equation
2//!
3//! A simple first-order ODE used to test numerical integrators:
4//!
5//! dx/dt = -k * x
6//!
7//! With k > 0, the analytical solution is: x(t) = x0 * exp(-k * t)
8//!
9//! This is a fundamental test case for ODE solvers, useful for studying
10//! stability and accuracy of numerical integration methods.
11//!
12//! The Dahlquist test equation is particularly useful for analyzing stiff solvers.
13
14use fmi::fmi3::{Fmi3Error, Fmi3Res};
15use fmi_export::fmi3::{Context, DefaultLoggingCategory, UserModel};
16use fmi_export::FmuModel;
17
18/// Dahlquist test equation model
19///
20/// This implements a simple first-order linear ODE: der(x) = -k * x
21#[derive(FmuModel, Default, Debug)]
22#[model(co_simulation = true, user_model = false)]
23pub struct Dahlquist {
24    /// Decay constant k
25    #[variable(causality = Parameter, start = 1.0, initial = Exact)]
26    pub k: f64,
27
28    /// State variable x
29    #[variable(causality = Output, start = 1.0, initial = Exact)]
30    pub x: f64,
31
32    /// Derivative of x (der_x = -k * x)
33    #[variable(causality = Local, derivative = x, initial = Calculated)]
34    der_x: f64,
35}
36
37impl UserModel for Dahlquist {
38    type LoggingCategory = DefaultLoggingCategory;
39
40    fn calculate_values(&mut self, _context: &dyn Context<Self>) -> Result<Fmi3Res, Fmi3Error> {
41        self.der_x = -self.k * self.x;
42        Ok(Fmi3Res::OK)
43    }
44
45    adml_solver::euler_cs_step!(0.001);
46}
47
48fmi_export::export_fmu!(Dahlquist);
49
50impl Dahlquist {
51    /// Create a new Dahlquist model with default parameters
52    pub fn new() -> Self {
53        Self {
54            k: 1.0,
55            x: 1.0,
56            der_x: -1.0,
57        }
58    }
59
60    /// Get the analytical solution at time t from initial value x0
61    pub fn analytical_solution(x0: f64, k: f64, t: f64) -> f64 {
62        x0 * (-k * t).exp()
63    }
64
65    /// Perform a single Euler integration step (for testing without FMI context)
66    pub fn do_step(&mut self, _current_time: f64, time_step: f64) {
67        self.der_x = -self.k * self.x;
68        self.x += self.der_x * time_step;
69    }
70}

72#[cfg(test)]
73mod tests {
74    use super::*;
75
76    #[test]
77    fn test_initial_values() {
78        let model = Dahlquist::new();
79        assert_eq!(model.x, 1.0);
80        assert_eq!(model.k, 1.0);
81    }
82
83    #[test]
84    fn test_derivative_calculation() {
85        let model = Dahlquist::new();
86
87        // Manually calculate what one step should do
88        let der_x = -model.k * model.x; // -1.0 * 1.0 = -1.0
89        assert_eq!(der_x, -1.0);
90    }
91
92    #[test]
93    fn test_analytical_solution() {
94        assert_eq!(Dahlquist::analytical_solution(1.0, 1.0, 0.0), 1.0);
95
96        let t1_value = Dahlquist::analytical_solution(1.0, 1.0, 1.0);
97        let expected = (-1.0f64).exp();
98        assert!((t1_value - expected).abs() < 1e-10);
99    }
100
101    #[test]
102    fn test_do_step() {
103        let mut model = Dahlquist::new();
104
105        let initial_x = model.x;
106        let dt = 0.1;
107
108        model.do_step(0.0, dt);
109
110        // After one Euler step: x_new = x + (-k*x)*dt = 1.0 + (-1.0)*0.1 = 0.9
111        assert!((model.x - 0.9).abs() < 1e-10);
112        assert!(model.x < initial_x); // Should decay
113    }
114}
